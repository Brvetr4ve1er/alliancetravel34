# Standard crawl policy: allow everything by default, except the CMS
# admin panel (/admin/), which must never be indexed.
# Asset directories (CSS/JS/images) are intentionally crawlable so
# Google Images can surface our hero photography.

User-agent: *
Allow: /
Disallow: /admin/

# Slow down the SEO crawlers that hammer the server. Add more here when
# the access logs show abuse.
#
# Disallow: /admin/ is repeated in each group ON PURPOSE. A group that names a
# user-agent REPLACES the "User-agent: *" group for that crawler — it does not
# add to it — so these two were being told "crawl at 5 s intervals" and nothing
# else, and the admin panel was fair game for exactly the two bots most likely
# to publish what they find.
User-agent: AhrefsBot
Crawl-delay: 5
Disallow: /admin/

User-agent: SemrushBot
Crawl-delay: 5
Disallow: /admin/

# Sitemap location — Google + Bing pick this up automatically.
